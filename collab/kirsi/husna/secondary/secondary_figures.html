Secondary figures: maternal diet and child metabolites

These figures show exposure-level detail, timing-window comparison, and robustness patterns for the age-specific and mutual-timing models. They are not the primary inferential summaries and should not be read as direct evidence for the warning-prone Q4 longitudinal model family unless stated explicitly.

1.7 Figure B7. Metabolite correlation heatmap

Show code 
mat_data <- as.matrix(df_all[, intersect(metabolite_cols, names(df_all)), drop = FALSE])
storage.mode(mat_data) <- "numeric"
keep_cols <- colSums(is.finite(mat_data)) >= 20
mat_data <- mat_data[, keep_cols, drop = FALSE]

if (ncol(mat_data) < 2) {
  ggplot() +
    annotate("text", x = 0, y = 0, label = "Metabolite correlation heatmap unavailable") +
    theme_void()
} else {
  cor_mat <- suppressWarnings(cor(mat_data, use = "pairwise.complete.obs"))
  cor_mat[!is.finite(cor_mat)] <- 0
  diag(cor_mat) <- 1
  cor_mat <- pmax(pmin(cor_mat, 1), -1)

  # Fetch NMR group metadata
  rd <- as.data.frame(rowData(experiments(mae)[["visit_all"]]), stringsAsFactors = FALSE)
  rd$Feature <- rownames(rd)
  rd$Group <- as.character(rd$Group)
  rd$Group[is.na(rd$Group) | rd$Group == ""] <- "Unknown"

  feature_meta_here <- rd %>%
    filter(Feature %in% colnames(cor_mat)) %>%
    arrange(match(Feature, colnames(cor_mat)))

  group_order <- unique(feature_meta_here$Group)
  
  ordered_names <- unlist(lapply(group_order, function(group_nm) {
    members <- feature_meta_here$Feature[feature_meta_here$Group == group_nm]
    if (length(members) > 2) {
      group_cor <- cor_mat[members, members, drop = FALSE]
      members[hclust(as.dist(1 - abs(group_cor)))$order]
    } else {
      members
    }
  }), use.names = FALSE)
  ordered_names <- intersect(ordered_names, colnames(cor_mat))

  feature_order <- data.frame(
    Feature = ordered_names,
    Position = seq_along(ordered_names),
    stringsAsFactors = FALSE
  ) %>%
    left_join(feature_meta_here[, c("Feature", "Group")], by = "Feature")

  group_ranges <- feature_order %>%
    group_by(Group) %>%
    summarise(
      start = min(Position) - 0.5,
      end = max(Position) + 0.5,
      center = mean(Position),
      n_features = n(),
      .groups = "drop"
    ) %>%
    mutate(Group = factor(Group, levels = group_order)) %>%
    arrange(Group)

  group_label_df <- group_ranges %>%
    mutate(Label = stringr::str_wrap(as.character(Group), width = 24))

  boundary_pos <- sort(unique(c(group_ranges$start, group_ranges$end)))
  boundary_pos <- boundary_pos[boundary_pos > 0.5 & boundary_pos < length(ordered_names) + 0.5]

  cor_long_df <- as.data.frame(cor_mat[ordered_names, ordered_names]) %>%
    mutate(Feature1 = rownames(cor_mat[ordered_names, ordered_names])) %>%
    pivot_longer(-Feature1, names_to = "Feature2", values_to = "Correlation") %>%
    left_join(feature_order %>% select(Feature1 = Feature, X = Position), by = "Feature1") %>%
    left_join(feature_order %>% select(Feature2 = Feature, Y = Position), by = "Feature2")

  ggplot(cor_long_df, aes(x = X, y = Y, fill = Correlation)) +
    geom_tile() +
    geom_vline(xintercept = boundary_pos, color = "grey45", linewidth = 0.18) +
    geom_hline(yintercept = boundary_pos, color = "grey45", linewidth = 0.18) +
    ggrepel::geom_text_repel(
      data = group_label_df,
      aes(x = 0.5, y = center, label = Label),
      inherit.aes = FALSE,
      nudge_x = -45,
      direction = "y",
      hjust = 1,
      size = 2.35,
      segment.color = "grey35",
      segment.size = 0.18,
      min.segment.length = 0,
      box.padding = 0.12,
      point.padding = 0.05,
      max.overlaps = Inf
    ) +
    scale_fill_gradient2(
      low = "#2166AC",
      mid = "white",
      high = "#B2182B",
      midpoint = 0,
      limits = c(-1, 1)
    ) +
    scale_x_continuous(limits = c(-70, length(ordered_names) + 0.5), expand = c(0, 0)) +
    scale_y_continuous(limits = c(0.5, length(ordered_names) + 0.5), expand = c(0, 0)) +
    coord_fixed(clip = "off") +
    labs(x = NULL, y = NULL, fill = "r", title = "Metabolite correlation structure") +
    theme_minimal(base_size = 12) +
    theme(
      panel.grid = element_blank(),
      axis.text.x = element_blank(),
      axis.text.y = element_blank(),
      axis.ticks = element_blank(),
      plot.margin = margin(5.5, 5.5, 5.5, 120),
      legend.position = "right"
    )
}
